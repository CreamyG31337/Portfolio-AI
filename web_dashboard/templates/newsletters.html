// "Blocked script execution" warning for every newsletter. The sandbox
// (no allow-scripts) is the real security control; this is cosmetic.
function sanitizeEmailHtml(html) {
    if (!hasRenderableHtml(html)) return '';
    return html
        .replace(/]*>[\s\S]*?<\/script>/gi, '')
        .replace(/]*\/>/gi, '');
}

async function viewEmail(id) {
    const modal = document.getElementById('email-view-modal');
    const loading = document.getElementById('email-modal-loading');
    const htmlView = document.getElementById('email-modal-html');
    const plainView = document.getElementById('email-modal-plain');
    const errorView = document.getElementById('email-modal-error');

    // Reset state
    loading.classList.remove('hidden');
    htmlView.classList.add('hidden');
    plainView.classList.add('hidden');
    errorView.classList.add('hidden');
    currentEmailData = null;

    // Show modal
    modal.classList.remove('hidden');
    modal.classList.add('flex'); // Flowbite modals use flex for centering
    modal.setAttribute('aria-modal', 'true');
    modal.setAttribute('role', 'dialog');
    document.body.style.overflow = 'hidden';

    try {
        const response = await fetch('/api/newsletters/' + id, { credentials: 'include' });
        if (!response.ok) throw new Error('HTTP ' + response.status);
        const data = await response.json();
        currentEmailData = data;

        // Populate header
        document.getElementById('email-modal-subject').textContent = data.subject || '(no subject)';
        document.getElementById('email-modal-from').textContent = data.sender_name || data.sender || 'Unknown';
        document.getElementById('email-modal-date').textContent = data.received_at
            ? new Date(data.received_at).toLocaleString()
            : '';

        loading.classList.add('hidden');

        const htmlAvailable = hasRenderableHtml(data.body_html);
        const plainAvailable = hasRenderablePlain(data.body_plain);

        // Show toggle buttons based on what's actually renderable
        const htmlBtn = document.getElementById('email-toggle-html');
        const plainBtn = document.getElementById('email-toggle-plain');
        htmlBtn.classList.toggle('hidden', !htmlAvailable);
        plainBtn.classList.toggle('hidden', !plainAvailable);

        // Default to HTML if available, else plain, else error
        if (htmlAvailable) {
            toggleEmailView('html');
        } else if (plainAvailable) {
            toggleEmailView('plain');
        } else {
            errorView.classList.remove('hidden');
            document.querySelector('#email-modal-error p').textContent = 'No email content available.';
        }
    } catch (err) {
        console.error('Error loading email:', err);
        loading.classList.add('hidden');
        errorView.classList.remove('hidden');
    }
}

function toggleEmailView(view) {
    if (!currentEmailData) return;

    const htmlView = document.getElementById('email-modal-html');
    const plainView = document.getElementById('email-modal-plain');
    const errorView = document.getElementById('email-modal-error');
    const htmlBtn = document.getElementById('email-toggle-html');
    const plainBtn = document.getElementById('email-toggle-plain');

    const htmlAvailable = hasRenderableHtml(currentEmailData.body_html);
    const plainAvailable = hasRenderablePlain(currentEmailData.body_plain);

    // If the user asked for a view we can't actually deliver, fall back gracefully
    // instead of writing "null" into the iframe (the historical bug).
    if (view === 'html' && !htmlAvailable) {
        view = plainAvailable ? 'plain' : 'none';
    } else if (view === 'plain' && !plainAvailable) {
        view = htmlAvailable ? 'html' : 'none';
    }

    currentEmailView = view;

    // Toggle button styles for Flowbite group
    if (view === 'html') {
        // HTML Active
        htmlBtn.className = "px-3 py-1.5 text-xs font-medium border border-accent rounded-s-lg focus:z-10 focus:ring-2 focus:ring-accent text-accent bg-dashboard-surface hover:bg-dashboard-surface-alt";
        plainBtn.className = "px-3 py-1.5 text-xs font-medium border-t border-b border-r border-border rounded-e-lg focus:z-10 focus:ring-2 focus:ring-accent text-text-secondary bg-dashboard-surface hover:bg-dashboard-surface-alt";

        // Render HTML in sandboxed iframe
        htmlView.classList.remove('hidden');
        plainView.classList.add('hidden');
        errorView.classList.add('hidden');

        const doc = htmlView.contentDocument || htmlView.contentWindow.document;
        doc.open();
        // TODO(ui-tailwind): Evaluate replacing this inline iframe style injection with a shared/minified Tailwind typography payload compatible with sandboxed newsletter rendering.
        // Resolve theme colors from the parent up front so the iframe doesn't have to
        // read parent CSS vars (it's same-origin today, but we want to keep the door
        // open to a tighter sandbox in the future).
        const parentStyles = window.getComputedStyle(document.documentElement);
        const colorText = (parentStyles.getPropertyValue('--text-primary') || '').trim() || '#e0e0e0';
        const colorBg = (parentStyles.getPropertyValue('--bg-secondary') || '').trim() || '#1a1a2e';
        const colorLink = (parentStyles.getPropertyValue('--color-link') || '').trim() || '#64b5f6';

        const safeHtml = sanitizeEmailHtml(currentEmailData.body_html);

        doc.write(`${safeHtml}`);

        doc.close();

        // Auto-resize iframe to content height (best-effort; same-origin sandbox only).
        const resize = function() {
            try {
                const h = htmlView.contentDocument.documentElement.scrollHeight;
                htmlView.style.minHeight = Math.max(400, h) + 'px';
            } catch (e) { /* iframe became cross-origin */ }
        };
        htmlView.onload = resize;
        setTimeout(resize, 100);

    } else if (view === 'plain') {
        // Plain Active
        htmlBtn.className = "px-3 py-1.5 text-xs font-medium border border-border rounded-s-lg focus:z-10 focus:ring-2 focus:ring-accent text-text-secondary bg-dashboard-surface hover:bg-dashboard-surface-alt";
        plainBtn.className = "px-3 py-1.5 text-xs font-medium border-t border-b border-r border-accent rounded-e-lg focus:z-10 focus:ring-2 focus:ring-accent text-accent bg-dashboard-surface hover:bg-dashboard-surface-alt";

        plainView.classList.remove('hidden');
        htmlView.classList.add('hidden');
        errorView.classList.add('hidden');
        plainView.textContent = currentEmailData.body_plain;
    } else {
        // Neither view is renderable -- show the error panel rather than a blank/null iframe.
        htmlView.classList.add('hidden');
        plainView.classList.add('hidden');
        errorView.classList.remove('hidden');
        document.querySelector('#email-modal-error p').textContent = 'No email content available.';
    }
}

function closeEmailModal() {
    const modal = document.getElementById('email-view-modal');
    modal.classList.add('hidden');
    modal.classList.remove('flex');
    modal.removeAttribute('aria-modal');
    modal.removeAttribute('role');
    document.body.style.overflow = '';
    currentEmailData = null;

    // Clear iframe
    const htmlView = document.getElementById('email-modal-html');
    try {
        const doc = htmlView.contentDocument || htmlView.contentWindow.document;
        doc.open();
        doc.write('');
        doc.close();
    } catch(e) {}
}

// Close modal on Escape key
document.addEventListener('keydown', function(e) {
    if (e.key === 'Escape') {
        const modal = document.getElementById('email-view-modal');
        if (!modal.classList.contains('hidden')) {
            closeEmailModal();
        }
    }
});
